  function setLanguage(lang) {
    currentLang = lang;

    if (lang === 'ar') {
      body.classList.remove('lang-fr');
      body.classList.add('lang-ar');
      html.setAttribute('dir', 'rtl');
      html.setAttribute('lang', 'ar');
    } else {
      body.classList.remove('lang-ar');
      body.classList.add('lang-fr');
      html.setAttribute('dir', 'ltr');
      html.setAttribute('lang', 'fr');
    }

    // Update all elements with data-i18n
    const elements = document.querySelectorAll('[data-i18n]');
    elements.forEach(el => {
      const key = el.getAttribute('data-i18n');
      if (translations[lang] && translations[lang][key]) {
        el.innerHTML = translations[lang][key];
      }
    });

    if (langName) langName.textContent = translations[lang].langBtn;
    if (langFlag) langFlag.textContent = translations[lang].langFlag;

  const cHours = document.getElementById('c-hours');

  let remainingSeconds = 3 * 3600 + 48 * 60 + 21;

      remainingSeconds--;

      remainingSeconds = 4 * 3600; // Reset urgency cycle
    }
    const h = Math.floor(remainingSeconds / 3600);
    const m = Math.floor((remainingSeconds % 3600) / 60);
    const s = remainingSeconds % 60;

    if (cHours) cHours.textContent = String(h).padStart(2, '0');
    if (cMins) cMins.textContent = String(m).padStart(2, '0');
    if (cSecs) cSecs.textContent = String(s).padStart(2, '0');
  }, 1000);

  // 3. Three.js 3D Interactive Blender Model

  const canvas = document.getElementById('blender-3d-canvas');
  let scene, camera, renderer, blenderGroup;
  let baseMesh, glassMesh, liquidMesh, bladeGroup;

  let previousMousePosition = { x: 0, y: 0 };

  const colorPalettes = {
    emerald: { base: 0x00A86B, accent: 0x00D084, name: 'أخضر زمردي فاخر (الأكثر طلباً)' },

  };

  function init3D() {
    if (!canvas || typeof THREE === 'undefined') {
      console.warn('Three.js not loaded, canvas will render high-res fallback.');
      return;
    }

    const width = canvas.parentElement.clientWidth || 500;
    const height = canvas.parentElement.clientHeight || 440;

    scene = new THREE.Scene();
    camera = new THREE.PerspectiveCamera(40, width / height, 0.1, 1000);
    camera.position.set(0, 0, 7.5);

    renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true, alpha: true });
    renderer.setSize(width, height);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

    // Studio Lights
    const ambientLight = new THREE.AmbientLight(0xffffff, 1.4);
    scene.add(ambientLight);

    const dirLight1 = new THREE.DirectionalLight(0xffffff, 2.0);
    dirLight1.position.set(5, 8, 5);
    scene.add(dirLight1);

    const dirLight2 = new THREE.DirectionalLight(0x00D084, 1.2);
    dirLight2.position.set(-5, -2, -3);
    scene.add(dirLight2);

    blenderGroup = new THREE.Group();

    // 1. Motor Base Cylinder
    const baseGeo = new THREE.CylinderGeometry(1.15, 1.15, 1.6, 32);
    const baseMat = new THREE.MeshStandardMaterial({
      color: colorPalettes.emerald.base,
      roughness: 0.35,
      metalness: 0.2
    });
    baseMesh = new THREE.Mesh(baseGeo, baseMat);
    baseMesh.position.y = -1.1;
    blenderGroup.add(baseMesh);

    // Metallic ring at bottom
    const ringGeo = new THREE.CylinderGeometry(1.16, 1.16, 0.2, 32);
    const ringMat = new THREE.MeshStandardMaterial({ color: 0xCBD5E1, roughness: 0.15, metalness: 0.85 });
    const botRing = new THREE.Mesh(ringGeo, ringMat);
    botRing.position.y = -1.8;
    blenderGroup.add(botRing);

    // Power Button on Base
    const btnGeo = new THREE.CylinderGeometry(0.24, 0.24, 0.1, 24);
    btnGeo.rotateX(Math.PI / 2);
    const btnMat = new THREE.MeshStandardMaterial({ color: 0xFFFFFF, roughness: 0.2, metalness: 0.6 });
    const powerBtn = new THREE.Mesh(btnGeo, btnMat);
    powerBtn.position.set(0, -1.05, 1.12);
    blenderGroup.add(powerBtn);

    // Glowing Power Ring
    const glowRingGeo = new THREE.TorusGeometry(0.28, 0.04, 16, 32);
    const glowRingMat = new THREE.MeshBasicMaterial({ color: 0x00D084 });
    const glowRing = new THREE.Mesh(glowRingGeo, glowRingMat);
    glowRing.position.set(0, -1.05, 1.14);
    blenderGroup.add(glowRing);

    // 2. Glass Cylinder Body
    const glassGeo = new THREE.CylinderGeometry(1.1, 1.1, 2.2, 32);
    const glassMat = new THREE.MeshPhysicalMaterial({
      color: 0xFFFFFF,
      transmission: 0.92,
      opacity: 1,
      transparent: true,
      roughness: 0.08,
      ior: 1.5,
      thickness: 0.4
    });
    glassMesh = new THREE.Mesh(glassGeo, glassMat);
    glassMesh.position.y = 0.8;
    blenderGroup.add(glassMesh);

    // 3. Green Smoothie Liquid inside
    const liqGeo = new THREE.CylinderGeometry(1.02, 1.02, 1.8, 32);
    const liqMat = new THREE.MeshStandardMaterial({
      color: 0x34D399,
      roughness: 0.4,
      metalness: 0.1,
      transparent: true,
      opacity: 0.85
    });
    liquidMesh = new THREE.Mesh(liqGeo, liqMat);
    liquidMesh.position.y = 0.7;
    blenderGroup.add(liquidMesh);

    // 4. Stainless Steel 6-Blade Mechanism inside
    bladeGroup = new THREE.Group();
    for (let i = 0; i < 3; i++) {
      const bladeGeo = new THREE.BoxGeometry(1.2, 0.06, 0.18);
      const bladeMat = new THREE.MeshStandardMaterial({ color: 0xE2E8F0, metalness: 0.95, roughness: 0.15 });
      const blade = new THREE.Mesh(bladeGeo, bladeMat);
      blade.rotation.y = (i * Math.PI) / 3;
      bladeGroup.add(blade);
    }
    bladeGroup.position.y = -0.2;
    blenderGroup.add(bladeGroup);

    // 5. Top Lid Cylinder
    const lidGeo = new THREE.CylinderGeometry(1.12, 1.12, 0.7, 32);
    const lidMat = new THREE.MeshStandardMaterial({
      color: colorPalettes.emerald.base,
      roughness: 0.35,
      metalness: 0.2
    });
    const topLid = new THREE.Mesh(lidGeo, lidMat);
    topLid.position.y = 2.25;
    blenderGroup.add(topLid);

    // Lid Metal Band
    const lidBandGeo = new THREE.CylinderGeometry(1.13, 1.13, 0.15, 32);
    const lidBand = new THREE.Mesh(lidBandGeo, ringMat);
    lidBand.position.y = 1.95;
    blenderGroup.add(lidBand);

    // Strap loop on top
    const strapGeo = new THREE.TorusGeometry(0.35, 0.08, 16, 24);
    strapGeo.rotateY(Math.PI / 2);
    const strapMat = new THREE.MeshStandardMaterial({ color: colorPalettes.emerald.accent, roughness: 0.8 });
    const strap = new THREE.Mesh(strapGeo, strapMat);
    strap.position.set(0.85, 2.35, 0);
    blenderGroup.add(strap);

    scene.add(blenderGroup);

    // Drag to rotate controls
    const wrapper = canvas.parentElement;

    wrapper.addEventListener('mousedown', (e) => {
      isDragging = true;
      previousMousePosition = { x: e.clientX, y: e.clientY };
    });

    window.addEventListener('mouseup', () => { isDragging = false; });

    wrapper.addEventListener('mousemove', (e) => {
      if (!isDragging || !blenderGroup) return;
      const deltaX = e.clientX - previousMousePosition.x;
      const deltaY = e.clientY - previousMousePosition.y;

      blenderGroup.rotation.y += deltaX * 0.012;
      blenderGroup.rotation.x += deltaY * 0.012;
      // Clamp x
      blenderGroup.rotation.x = Math.max(-0.6, Math.min(0.6, blenderGroup.rotation.x));

      previousMousePosition = { x: e.clientX, y: e.clientY };
    });

    // Touch support for mobile devices
    wrapper.addEventListener('touchstart', (e) => {
      if (e.touches.length === 1) {
        isDragging = true;
        previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY };
      }
    }, { passive: true });

    window.addEventListener('touchend', () => { isDragging = false; });

    wrapper.addEventListener('touchmove', (e) => {
      if (!isDragging || !blenderGroup || e.touches.length !== 1) return;
      const deltaX = e.touches[0].clientX - previousMousePosition.x;
      const deltaY = e.touches[0].clientY - previousMousePosition.y;

      blenderGroup.rotation.y += deltaX * 0.012;
      blenderGroup.rotation.x += deltaY * 0.012;
      blenderGroup.rotation.x = Math.max(-0.6, Math.min(0.6, blenderGroup.rotation.x));

      previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY };
    }, { passive: true });

    // Animation Loop
    function animate() {
      requestAnimationFrame(animate);

      if (!isDragging && blenderGroup) {
        blenderGroup.rotation.y += 0.005; // Gentle continuous auto-spin
      }

      if (bladeGroup) {
        bladeGroup.rotation.y += 0.05; // Spin blades gently
      }

      renderer.render(scene, camera);
    }
    animate();

    // Responsive Canvas Resize
    window.addEventListener('resize', () => {
      if (!canvas || !renderer || !camera) return;
      const newW = canvas.parentElement.clientWidth;
      const newH = canvas.parentElement.clientHeight;
      camera.aspect = newW / newH;
      camera.updateProjectionMatrix();
      renderer.setSize(newW, newH);
    });
  }

        return;

      }
    });
  });
